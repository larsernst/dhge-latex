% ROOT & PACKAGE SETUP
\documentclass[a4paper, 12pt]{article}

% Präambel laden

\input{../preamble.tex}

% Nutzerkonfiguration laden
\input{../config.tex}

% verwendete Pakete laden und konfigurieren
\input{components/package.config.tex}

% Setup von Commands und Dokument
\input{components/commands.tex}
% => Das LaTeX Arbeitsverzeichnis ist das der Haupt-Datei, also build/template.tex => build/

\input{components/setup.tex}
% ab hier ist das Arbeitsverzeichnis das root Verzeichnis (gilt primär für \input)

% Glossar-Einträge (CGLOSSAR=1) bzw. Abkürzungen (CGLOSSAR=0) müssen früher
% eingefügt werden, da sie nicht als Text, sondern als Variablen definiert werden
\if\CGLOSSAR 1
    \input{glossar.tex}
\else
    \input{abk.tex}
\fi


% Konfiguration globaler Definitionen

% PDF Metadata
\hypersetup{
    pdftitle={\CTITLE},
    pdfauthor={\CAUTHOR}
}

\addbibresource	{../literatur.bib}

% Hier beginnt der Spaß
% setzt variablen für den Titel -> \maketitle in deckblatt.tex
\title{{\LARGE \textbf{\CTITLE}}}
\author{}
\date{}


\begin{document}
\pagenumbering{gobble}
    % lade Deckblatt ohne Nummerierung
    \input{build/components/deckblatt.tex}

    % Lade Sperrvermerk abhängig von der Einstellung in config.tex
    \if\CSPERRVERMERK 1
        \input{build/components/sperrvermerk.tex}
    \fi

    % Lade Thesenblatt abhängig von der Einstellung in config.tex
    \if\CARBEIT B
        \input{../bachelor.tex}
    \fi

    % Lade Abstract abhängig von der Einstellung in config.tex
    % abstract.tex enthält die deutsche Kurzfassung und das englische Abstract
    % (jeweils 2 Absätze), beides hinter dem Deckblatt und vor dem Inhaltsverzeichnis
    \if\CHASABSTRACT 1
        \input{../abstract.tex}
        \newpage % Sections haben hier noch keinen automatischen Seitenumbruch
    \fi

    % INHALTSVERZEICHNIS
    \pagenumbering{Roman} \setcounter{page}{1}
    \tableofcontents{\fancyfoot{}}
    % im Kusche-Mode erscheinen die Anhänge als normale Hauptkapitel
    % (A, B, C, ...) am Ende des Inhaltsverzeichnisses, es gibt KEIN separates
    % Anhangs-Verzeichnis (siehe \anhang in anlagen.tex)
    \cleardoublepage

    % ABBILDUNGSVERZEICHNIS
    \phantomsection
    \conditionalLoF

    % TABELLENVERZEICHNIS
    \phantomsection
    \conditionalLoT

    % ABKÜRZUNGSVERZEICHNIS (nur im klassischen Modus CGLOSSAR=0)
    % im Glossar-Modus (CGLOSSAR=1) erscheint das Glossar hinter den Anhängen
    % und vor dem Literaturverzeichnis (siehe weiter unten im Dokument)
    % wird nur dann generiert, wenn mindestens ein mal \ac oder ein verwandter Befehl aufgerufen wurde und damit nicht leer ist
    \if\CGLOSSAR 0
        \ifnum\totvalue{acro_num}>0
            \printacronyms
            \addcontentsline{toc}{section}{Abkürzungsverzeichnis}
            \cleardoublepage
        \fi
    \fi

    \setcounter{savepage}{\arabic{page}}

    % MAIN CONTENT
    \pagenumbering{arabic}
    \input{main.tex}
    \cleardoublepage

    % LITERATURVERZEICHNIS
    % TODO Formatierung

    % Im Kusche-Mode wird arabische Nummerierung beibehalten
    \if\CKUSCHE 0
        \pagenumbering{Roman} \setcounter{page}{\thesavepage}
    \fi

    % wenn keine Literatur verwendet (zitiert) wird, erstelle kein Literaturverzeichnis
    % im Kusche-Mode wird das LiteraturVZ später generiert
    \if\CKUSCHE 0
        \ifnum\thetotalbibentries>0
            \printbibliography[title=Literaturverzeichnis]
            \addcontentsline{toc}{section}{Literaturverzeichnis}
            \cleardoublepage
        \fi
    \fi


    % TODO %ANLAGENVERZEICHNIS UND ANLAGEN
    % erzeugt ein Anlagenkapitel und fügt es zum InhaltsVZ hinzu, insofern Anlagen existieren.

    % \listofanlagen generiert ein Anlagenverzeichnis
    % im Kusche-Modus wird kein separates Anlagenverzeichnis erzeugt: die Anhänge
    % werden mit \anhang{...} als normale Hauptkapitel (A, B, C, ...) am Ende des
    % Inhaltsverzeichnisses geführt
    \ifnum\totvalue{anlagenentries}>0
    \section*{Anlagen}
    \addcontentsline{toc}{section}{Anlagen}
    \if\CKUSCHE 0
        \begin{spacing}{2}
            \listofanlagen
        \end{spacing}
        \cleardoublepage
    \fi
    \fi

    % deaktiviere Kapitel in Fußzeile im Kusche Mode ab hier
    \if\CKUSCHE 1
        \fancyfoot      [L]{}
    \fi

    % Anlagen/Anhänge:
    % im Kusche-Modus wird \renewFigTabCap nicht benötigt, die Abbildungen und
    % Tabellen in den Anhängen werden über \anhang{...} (Anhang A, B, C, ...)
    % automatisch zweistufig (A.1, A.2, B.1, ...) nummeriert
    \if\CKUSCHE 0
        \renewFigTabCap % Verhalten von Figure und Table Environments sowie \caption wird verändert (TODO: wie genau? LG, ZPM)
    \fi
    \input{anlagen.tex}
    \cleardoublepage

    % GLOSSAR (nur im Glossar-Modus CGLOSSAR=1)
    % Das Glossar ersetzt das Abkürzungsverzeichnis und gehört
    % hinter die Anhänge, aber vor das Literaturverzeichnis.
    % Einträge werden in glossar.tex definiert.
    \if\CGLOSSAR 1
        \phantomsection
        \printglossary[title=Glossar]
        \addcontentsline{toc}{section}{Glossar}
        \cleardoublepage
    \fi

    % Im Kusche-Mode kommt das LiteraturVZ zuletzt
    \if\CKUSCHE 1
        \ifnum\thetotalbibentries>0
            \printbibliography[title=Literaturverzeichnis]
            \addcontentsline{toc}{section}{Literaturverzeichnis}
            \cleardoublepage
        \fi
    \fi
    
    % Einrückung und Abstand unabhängig von config.tex setzen: Ehrenwort und Freigabe korrekt formatiert
    \setlength{\parskip}{6pt}
    \setlength{\parindent}{0cm}

    % EHRENWÖRTLICHE ERKLÄRUNG

    \pagestyle{empty}
    \pagenumbering{gobble}
    \section*{Eigenständigkeitserklärung}
    \addcontentsline{toc}{section}{Eigenständigkeitserklärung}
    \input{build/components/ehrenwort.tex}

    \if\CARBEIT B
        \input{components/freigabe.tex}
    \fi

\end{document}
